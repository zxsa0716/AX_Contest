# ── D. ERA5 기상변수 추출 ───────────────────────────────────────────── def get_monthly_era5(lat, lon, company_id, buffer_m=10000): """풍속·강수·기온·경계층 높이 월별 평균 추출""" point = ee.Geometry.Point([lon, lat]) roi = point.buffer(buffer_m) era5 = ee.ImageCollection('ECMWF/ERA5_LAND/HOURLY') results = [] for year in range(2019, 2024): for month in range(1, 13): start = f'{year}-{month:02d}-01' end_m = month + 1 if month < 12 else 1 end_y = year if month < 12 else year + 1 end = f'{end_y}-{end_m:02d}-01' monthly = (era5 .filterDate(start, end) .filterBounds(roi) .select([ 'u_component_of_wind_10m', # u10: 동-서 풍속 'v_component_of_wind_10m', # v10: 남-북 풍속 'total_precipitation_hourly', # tp: 강수량 'temperature_2m', # t2m: 2m 기온 'boundary_layer_height' # blh: 경계층 높이 ]) .mean()) stats = monthly.reduceRegion( reducer=ee.Reducer.mean(), geometry=roi, scale=11000 # ERA5 네이티브 해상도 ~11km ).getInfo() # 풍속 크기 계산: sqrt(u^2 + v^2) u = stats.get('u_component_of_wind_10m', 0) v = stats.get('v_component_of_wind_10m', 0) wind_speed = (u**2 + v**2)**0.5 if u and v else None results.append({ 'company_id': company_id, 'year': year, 'month': month, 'wind_speed_ms': wind_speed, 'u10': u, 'v10': v, 'precip_m': stats.get('total_precipitation_hourly'), 'temp_k': stats.get('temperature_2m'), 'blh_m': stats.get('boundary_layer_height') }) return pd.DataFrame(results)

DART API 재무정보 수집 ──────────────────────────────────────── import OpenDartReader as odr # pip install opendartreader import requests, zipfile from io import BytesIO import xmltodict DART_KEY = "YOUR_DART_API_KEY" dart = odr.OpenDartReader(DART_KEY) # ── STEP 1: 전체 기업 고유번호 목록 다운로드 ──────────────────────── url = f"https://opendart.fss.or.kr/api/corpCode.xml?crtfc_key={DART_KEY}" res = requests.get(url) corp_zip = zipfile.ZipFile(BytesIO(res.content)) corp_xml = corp_zip.read('CORPCODE.xml') corp_dict = xmltodict.parse(corp_xml) corp_df = pd.DataFrame(corp_dict['result']['list']) # corp_code: DART 고유번호, stock_code: 종목코드(상장사만 있음), corp_name: 기업명 listed = corp_df[corp_df['stock_code'].notna()] # 상장사만 필터 print(f"상장사 수: {len(listed)}") # ── STEP 2: 코스피 200 종목코드 → corp_code 매핑 ──────────────────── # 코스피 200 구성종목: KRX 홈페이지(krx.co.kr) → 지수 → 코스피200 → 구성종목 Excel 다운로드 kospi200 = pd.read_excel("data/KOSPI200_종목.xlsx") kospi200 = kospi200.merge(listed, left_on='종목코드', right_on='stock_code', how='left') # ── STEP 3: 연도별 재무제표 수집 ───────────────────────────────────── fin_data = [] for _, row in kospi200.iterrows(): corp_code = row['corp_code'] if pd.isna(corp_code): continue for year in range(2019, 2024): try: # 연결재무제표 우선, 없으면 개별 fs = dart.finstate_all(corp_code, year, 'CFS') # CFS: 연결, OFS: 개별 if fs is None or len(fs) == 0: fs = dart.finstate_all(corp_code, year, 'OFS') if fs is not None: # 매출액, 자산총계 추출 revenue = fs[fs['account_nm'].str.contains('매출액|수익', na=False)] assets = fs[fs['account_nm'].str.contains('자산총계|자산 합계', na=False)] fin_data.append({ 'corp_code': corp_code, 'corp_name': row['corp_name'], 'stock_code': row['stock_code'], 'year': year, 'revenue': revenue['thstrm_amount'].iloc[0] if len(revenue) > 0 else None, 'total_assets': assets['thstrm_amount'].iloc[0] if len(assets) > 0 else None }) except Exception as e: print(f"Error {corp_code} {year}: {e}") continue time.sleep(0.3) # rate limit fin_df = pd.DataFrame(fin_data)

# ── 통합 매칭 파이프라인 ────────────────────────────────────────────── from rapidfuzz import fuzz, process import re # ── 1단계: 사업자번호 직접 매칭 (오차 0) ───────────────────────────── def normalize_bizno(bizno): """하이픈 제거, 10자리 숫자 표준화""" if pd.isna(bizno): return None return re.sub(r'[^0-9]', '', str(bizno)).zfill(10) gir['bizno_clean'] = gir['사업자등록번호'].apply(normalize_bizno) dart_fin['bizno_clean'] = dart_fin['bizr_no'].apply(normalize_bizno) merged_direct = gir.merge(dart_fin, on='bizno_clean', how='inner') print(f"직접 매칭 성공: {len(merged_direct['bizno_clean'].unique())}개사") unmatched_gir = gir[~gir['bizno_clean'].isin(merged_direct['bizno_clean'])] # ── 2단계: 기업명 fuzzy 매칭 (직접 매칭 실패 기업 대상) ────────────── def fuzzy_match_company(query, choices, threshold=85): result = process.extractOne( query, choices, scorer=fuzz.token_sort_ratio, score_cutoff=threshold ) return result[0] if result else None dart_names = dart_fin['corp_name'].unique().tolist() fuzzy_results = [] for _, row in unmatched_gir.iterrows(): match = fuzzy_match_company(row['업체명'], dart_names) if match: score = fuzz.token_sort_ratio(row['업체명'], match) fuzzy_results.append({ 'gir_name': row['업체명'], 'dart_match': match, 'score': score, 'needs_review': score < 90 # 90 미만은 수동 검수 플래그 }) fuzzy_df = pd.DataFrame(fuzzy_results) # 수동 검수 필요 목록 저장 fuzzy_df[fuzzy_df['needs_review']].to_csv('manual_review.csv', index=False) print(f"수동 검수 필요: {fuzzy_df['needs_review'].sum()}건")
